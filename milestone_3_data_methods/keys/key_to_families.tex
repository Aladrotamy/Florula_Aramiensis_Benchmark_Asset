\chapter{Key to Families}
\label{ch:key-families}

This key covers only the eight families treated in this volume. Where
a family keys out twice, the genus reached at that lead is given in
brackets.

\begin{florakey}
	\lead{1a}{Leaves compound}{2}
	\lead{1b}{Leaves simple}{5}
	\lead{2a}{Leaves digitately compound, the leaflets all arising from
		the tip of the petiole}{Malvaceae (\taxon{Adansonia}, \taxon{Ceiba})}
	\lead{2b}{Leaves pinnate or bipinnate}{3}
	\lead{3a}{Fruit a woody capsule splitting into 4 or 5 valves; seeds
		winged}{Meliaceae}
	\lead{3b}{Fruit not a valved capsule}{4}
	\lead{4a}{Leaves once pinnate with a terminal leaflet; crushed leaves
		with a resinous smell; fruit a fleshy drupe}{Anacardiaceae
		(\taxon{Spondias})}
	\lead{4b}{Leaves bipinnate, or pinnate with a swollen pulvinus at the
		base of the petiole; fruit a pod or a flat winged disc}{Fabaceae}
	\lead{5a}{Slash exuding white latex}{6}
	\lead{5b}{Slash without white latex}{7}
	\lead{6a}{Stipules present, leaving a scar round the twig; flowers
		small, crowded in spikes or heads}{Moraceae}
	\lead{6b}{Stipules absent; leaves crowded at the ends of the twigs;
		fruit a berry with large glossy seeds}{Sapotaceae}
	\lead{7a}{Terminal bud enclosed in a long conical stipule that leaves
		a ring scar; fruit a large drupe}{Irvingiaceae}
	\lead{7b}{Terminal bud not enclosed in a conical stipule}{8}
	\lead{8a}{Branches in tiers, with the leaves crowded at their ends;
		fruit dry and winged}{Combretaceae}
	\lead{8b}{Branches not in tiers; fruit not winged}{9}
	\lead{9a}{Fruit a kidney-shaped nut borne on a swollen, fleshy
		stalk}{Anacardiaceae (\taxon{Anacardium})}
	\lead{9b}{Fruit of 4 or 5 spreading follicles with large seeds;
		petiole swollen at both ends}{Malvaceae (\taxon{Cola})}
\end{florakey}
